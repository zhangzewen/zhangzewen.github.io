kubernetes 源码分析之 apiserver admission

Created: Apr 14, 2025 · 5 min read

本篇文章基于 apiserver v0.25.1

准入控制（ admission control ）会在请求经过认证和授权前置检查后，资源对象被持久化（存储到 etcd ）前的最后一道防线，准入控制会对资源对象进行修改或者验证。apiserver 准入控制由一系列的插件组成，同时也提供了 webhook 准入控制插件支持用户自定义的准入控制插件来进行扩展。

接口定义

admission plugins 根据功能会实现如下接口：

// k8s.io/apiserver/pkg/admission/interfaces.go
type Interface interface {
	Handles(operation Operation) bool
}

type MutationInterface interface {
	Interface
	Admit(ctx context.Context, a Attributes, o ObjectInterfaces) (err error)
}

type ValidationInterface interface {
	Interface
	Validate(ctx context.Context, a Attributes, o ObjectInterfaces) (err error)
}

Interface 是基础的接口，其方法 Handles 会判断是是否可以处理操作某个资源的动作：

const (
Create  Operation = "CREATE"
Update  Operation = "UPDATE"
Delete  Operation = "DELETE"
Connect Operation = "CONNECT"

接口 MutationInterface 和 ValidationInterface 扩展了 Interface 接口，前者可能会修改传入的资源对象，后者不会修改资源对象，但是会对资源对象进行校验。

admission plugins 管理

注册

apiserver 使用结构体 Plugins 实现 admission plugins 管理：

// k8s.io/apiserver/pkg/admission/plugins.go
type Plugins struct {
	lock     sync.Mutex
	registry map[string]Factory
}

可以看到插件的管理很简单，就是一个映射表的管理，key 是插件名字，value 是 Factory，其作用是生成 plugin 实例：

type Factory func(config io.Reader) (Interface, error)

通过 Register 方法来注册插件：

// k8s.io/apiserver/pkg/admission/plugins.go
func (ps *Plugins) Register(name string, plugin Factory) {
    ps.lock.Lock()
    defer ps.lock.Unlock()
    if ps.registry != nil {
        _, found := ps.registry[name]
        if found { //如果重复注册，直接fatal
            klog.Fatalf("Admission plugin %q was registered twice", name)
        }
    } else {//惰性创建map
        ps.registry = map[string]Factory{}
    }

    klog.V(1).InfoS("Registered admission plugin", "plugin", name)
    //注册plugin
    ps.registry[name] = plugin
}

创建 admission plugin 实例

通过调用 InitPlugin 来创建 admission plugin 实例：

// k8s.io/apiserver/pkg/admission/plugins.go
func (ps *Plugins) InitPlugin(name string, config io.Reader, pluginInitializer PluginInitializer) (Interface, error) {
    if name == "" {
        klog.Info("No admission plugin specified.")
        return nil, nil
    }
    // 创建plugin
    plugin, found, err := ps.getPlugin(name, config)
    if err != nil {//出错了
        return nil, fmt.Errorf("couldn't init admission plugin %q: %v", name, err)
    }
    if !found { //没有找到
        return nil, fmt.Errorf("unknown admission plugin: %s", name)
    }
    //初始化和初始化校验逻辑
    pluginInitializer.Initialize(plugin)
    // ensure that plugins have been properly initialized
    // 校验初始化是否合法， 函数的实现在下面
    if err := ValidateInitialization(plugin); err != nil {
        return nil, fmt.Errorf("failed to initialize admission plugin %q: %v", name, err)
    }

    return plugin, nil
}
func ValidateInitialization(plugin Interface) error {
    // 查看plugin是否实现了InitializationValidator接口，如果实现了，调用即可
    if validater, ok := plugin.(InitializationValidator); ok {
        err := validater.ValidateInitialization()
        if err != nil {
            return err 
        }
    }
    return nil 
}

在上面的代码中调用方法 getPlugin 创建 admission plugin 实例，然后进行初始化和初始化校验，可以结合initializer来详看：

func (ps *Plugins) getPlugin(name string, config io.Reader) (Interface, bool, error) {
    ps.lock.Lock()
    defer ps.lock.Unlock()
    f, found := ps.registry[name]
    if !found {//没有被注册，直接返回
        return nil, false, nil
    }
    //把参数config表示的I/O流读入到config1， config2， config1==config2
    config1, config2, err := splitStream(config)
    if err != nil {
        return nil, true, err
    }
  
    if !PluginEnabledFn(name, config1) { //PluginEnabledFn这个返回true，可以代码，这个是写死了的
        return nil, true, nil
    }
    // 通过factory来构造Interface实例
    ret, err := f(config2)
    return ret, true, err
}

通过 NewFromPlugins 可以创建下文提到的 admission plugins 调用链：

func (ps *Plugins) NewFromPlugins(pluginNames []string, configProvider ConfigProvider, pluginInitializer PluginInitializer, decorator Decorator) (Interface, error) {
    handlers := []Interface{}
    mutationPlugins := []string{}
    validationPlugins := []string{}
    for _, pluginName := range pluginNames {
        pluginConfig, err := configProvider.ConfigFor(pluginName)
        if err != nil {
            return nil, err
        }
        //创建plugin实例
        plugin, err := ps.InitPlugin(pluginName, pluginConfig, pluginInitializer)
        if err != nil {
            return nil, err
        }
        if plugin != nil {
            if decorator != nil {//有装饰器
                handlers = append(handlers, decorator.Decorate(plugin, pluginName))
            } else {//没有装饰器
                handlers = append(handlers, plugin)
            }

            if _, ok := plugin.(MutationInterface); ok {
                mutationPlugins = append(mutationPlugins, pluginName)
            }
            if _, ok := plugin.(ValidationInterface); ok {
                validationPlugins = append(validationPlugins, pluginName)
            }
        }
    }
    if len(mutationPlugins) != 0 {
        klog.Infof("Loaded %d mutating admission controller(s) successfully in the following order: %s.", len(mutationPlugins), strings.Join(mutationPlugins, ","))
    }
    if len(validationPlugins) != 0 {
        klog.Infof("Loaded %d validating admission controller(s) successfully in the following order: %s.", len(validationPlugins), strings.Join(validationPlugins, ","))
    }
    return newReinvocationHandler(chainAdmissionHandler(handlers)), nil
}

初始化器：initializer

kubernetes 提供了admission plugin 初始化验证的基础接口和五个扩展接口以扩展不同的能力：

  • WantsExternalKubeClientSet：扩展了其设置和 apiserver 交互的客户端能力。
  • WantsExternalKubeInformerFactory：扩展了其使用 informer 的能力。
  • WantsAuthorizer：扩展了使用 authorize 的能力。
  • WantsQuotaConfiguration：扩展了使用 quota 配置的能力。
  • WantsFeatures：扩展了其知道 feature 是否启用的能力。
// k8s.io/apiserver/pkg/admission/interfaces.go
type InitializationValidator interface {
    ValidateInitialization() error
}


// k8s.io/apiserver/pkg/admission/initializer/interfaces.go
type WantsExternalKubeClientSet interface {
    SetExternalKubeClientSet(kubernetes.Interface)
    admission.InitializationValidator
}

type WantsExternalKubeInformerFactory interface {
    SetExternalKubeInformerFactory(informers.SharedInformerFactory)
    admission.InitializationValidator
}

type WantsAuthorizer interface {
    SetAuthorizer(authorizer.Authorizer)
    admission.InitializationValidator
}

type WantsQuotaConfiguration interface {
    SetQuotaConfiguration(quota.Configuration)
    admission.InitializationValidator
}

type WantsFeatures interface {
    InspectFeatureGates(featuregate.FeatureGate)
    admission.InitializationValidator
}

结构体 pluginInitializer，是一个默认的初始化器，它实现了接口 PluginInitializer 。可以看到默认的初始化器 pluginInitializer 的字段正好对应上了上述的初始化验证器的4个扩展接口。

// k8s.io/apiserver/pkg/admission/initializer/initializer.go
type pluginInitializer struct {
    externalClient    kubernetes.Interface
    externalInformers informers.SharedInformerFactory
    authorizer        authorizer.Authorizer
    featureGates      featuregate.FeatureGate
}
func New(
    extClientset kubernetes.Interface,
    extInformers informers.SharedInformerFactory,
    authz authorizer.Authorizer,
    featureGates featuregate.FeatureGate,
) pluginInitializer {
    return pluginInitializer{
        externalClient:    extClientset,
        externalInformers: extInformers,
        authorizer:        authz,
        featureGates:      featureGates,
    }
}

pluginInitializer 实现接口 PluginInitializer 的方法 Initialize，该方法接受接口 Interface 实例，即 admission plugin 实例，根据其实现的四个接口的个数来赋予其不同的能力，且 admission plugin 必须要实现接口 InitializationValidator，即在初始化完成后，需要调用 InitializationValidator 中的方法 ValidateInitialization 验证初始化完成。

func (i pluginInitializer) Initialize(plugin admission.Interface) {
    if wants, ok := plugin.(WantsFeatures); ok {
        wants.InspectFeatureGates(i.featureGates)
    }

    if wants, ok := plugin.(WantsExternalKubeClientSet); ok {
        wants.SetExternalKubeClientSet(i.externalClient)
    }

    if wants, ok := plugin.(WantsExternalKubeInformerFactory); ok {
        wants.SetExternalKubeInformerFactory(i.externalInformers)
    }

    if wants, ok := plugin.(WantsAuthorizer); ok {
        wants.SetAuthorizer(i.authorizer)
    }
}

admission plugins 调用链

kubernetes 把 admission plugins 都聚合在一个数组中，通过遍历该数组来依次调用 admission plugin：

// k8s.io/apiserver/pkg/admission/chain.go
type chainAdmissionHandler []Interface

func NewChainHandler(handlers ...Interface) chainAdmissionHandler {
    return chainAdmissionHandler(handlers)
}

且 chainAdmissionHandler 实现了接口定义中所述的三个接口:

func (admissionHandler chainAdmissionHandler) Admit(ctx context.Context, a Attributes, o ObjectInterfaces) error {
    for _, handler := range admissionHandler {
        // 如果不支持操作某个动作(CREATE/UPDATED/DELETE/CONNECT),直接忽略，然后继续遍历循环
        if !handler.Handles(a.GetOperation()) {
            continue
        }
        // 是否实现了MutationInterface接口
        if mutator, ok := handler.(MutationInterface); ok {
            err := mutator.Admit(ctx, a, o)
            if err != nil {
                return err
            }
        }
    }
    return nil
}

func (admissionHandler chainAdmissionHandler) Validate(ctx context.Context, a Attributes, o ObjectInterfaces) error {
    for _, handler := range admissionHandler {
        // 如果不支持操作某个动作(CREATE/UPDATED/DELETE/CONNECT),直接忽略，然后继续遍历循环
        if !handler.Handles(a.GetOperation()) {
            continue
        }
        // 是否实现了ValidationInterface接口
        if validator, ok := handler.(ValidationInterface); ok {
            err := validator.Validate(ctx, a, o)
            if err != nil {
                return err
            }
        }
    }
    return nil
}

// 遍历判断是否支持某个动作(CREATE/UPDATED/DELETE/CONNECT)， 只要有一个plugin支持，就返回true，如果全部都不支持，就返回false
func (admissionHandler chainAdmissionHandler) Handles(operation Operation) bool {
    for _, handler := range admissionHandler {
        if handler.Handles(operation) {
            return true
        }
    }
    return false
}

像类似 restfulCreateNamedResource 这样签名的函数在 registerResourceHandlers 中还有很多，即当对 Resource 进行创建/删除/更新时候，admission plugins 会对这些操作进行拦截，执行 Mutation/Validation 逻辑，然后再进行持久化。需要注意的是 admission plugins 不会作用于读操作上，具体可以看看 case "GET", case "LIST", case "WATCH", case "WATCHLIST" 这几个分支。 然后就是 admit 是怎么来的， 请看调用链：
