kubernetes 源码分析之 apiserver storage etcd 层实现分析

Created: May 11, 2025 · 5 min read

本篇文章基于 apiserver v0.25.1

kubernetes 的存储后端除了 etcd 也可以支持别的存储后端，但都需要实现 Interface 接口：

  • Versioner： 资源对象的 metadata 操作（更新 resourceversion 和 selflink ）接口，结构体 APIObjectVersioner 实现了该接口。
  • Create 创建资源对象的方法。
  • Delete 删除资源对象的方法。
  • Watch 通过Watch机制监控资源对象变化方法。
  • Get 获取资源对象的方法。
  • GetList 获取资源对象的方法，以列表的形式返回。
  • GuaranteedUpdate： 保证传入的tryUpdate函数运行成功。
  • Count： 获取指定key下的条目数量。
type Interface interface {
	Versioner() Versioner
	Create(ctx context.Context, key string, obj, out runtime.Object, ttl uint64) error
	Delete(
		ctx context.Context, key string, out runtime.Object, preconditions *Preconditions,
		validateDeletion ValidateObjectFunc, cachedExistingObject runtime.Object) error
	Watch(ctx context.Context, key string, opts ListOptions) (watch.Interface, error)
	Get(ctx context.Context, key string, opts GetOptions, objPtr runtime.Object) error
	GetList(ctx context.Context, key string, opts ListOptions, listObj runtime.Object) error
	GuaranteedUpdate(
		ctx context.Context, key string, destination runtime.Object, ignoreNotFound bool,
		preconditions *Preconditions, tryUpdate UpdateFunc, cachedExistingObject runtime.Object) error
	Count(key string) (int64, error)
}

在目录 k8s.io/apiserver/pkg/storage/etcd3 中的代码对 etcd 官方库进行了封装， 结构体 store 实现了该接口，以 Get 为例：

  • 通过官方库获取 etcd 集群中资源对象的数据。
  • 解码从 etcd 获取的二进制数据，解码后的数据存储在objPtr。
  • 最后通过 versioner.UpdateObject 函数更新最新的资源对象的 resourceVersion 资源版本号(即 kv.ModRevision， 也称为 modifiedIndex)。
// Get implements storage.Interface.Get.
func (s *store) Get(ctx context.Context, key string, opts storage.GetOptions, out runtime.Object) error {
	key = path.Join(s.pathPrefix, key)
	startTime := time.Now()
    // 使用官方库获取 key 所对应的资源
	getResp, err := s.client.KV.Get(ctx, key)
	metrics.RecordEtcdRequestLatency("get", getTypeName(out), startTime)
	if err != nil {
		return err
	}
	if err = s.validateMinimumResourceVersion(opts.ResourceVersion, uint64(getResp.Header.Revision)); err != nil {
		return err
	}
    // 处理记录不存在的情况
	if len(getResp.Kvs) == 0 {
		if opts.IgnoreNotFound {
			return runtime.SetZeroValue(out)
		}
		return storage.NewKeyNotFoundError(key, 0)
	}
	kv := getResp.Kvs[0]
    // 解码并存储在参数 out 中，并更新 资源版本号
	data, _, err := s.transformer.TransformFromStorage(ctx, kv.Value, authenticatedDataString(key))
	if err != nil {
		return storage.NewInternalError(err.Error())
	}

	return decode(s.codec, s.versioner, data, out, kv.ModRevision)
}

func decode(codec runtime.Codec, versioner storage.Versioner, value []byte, objPtr runtime.Object, rev int64) error {
	if _, err := conversion.EnforcePtr(objPtr); err != nil {
		return fmt.Errorf("unable to convert output object to pointer: %v", err)
	}
	_, _, err := codec.Decode(value, nil, objPtr)
	if err != nil {
		return err
	}
	// being unable to set the version does not prevent the object from being extracted
	if err := versioner.UpdateObject(objPtr, uint64(rev)); err != nil {
		klog.Errorf("failed to update object version: %v", err)
	}
	return nil
}

Get 返回的 reversion 是 etcd 中的 mod reversion。 但是GetList的就不一样了, 是etcd返回的结果中的header中的reversion， 这个reversion是全局的，是单调递增的值。这个reversion是单个key的create_reversion和mod_reversion的来源。

watch机制的实现

结构体 watchChan 实现了接口 watch 机制的接口 Interface：

// k8s.io/apimachinery/pkg/watch/watch.go
type Interface interface {
	Stop()
	ResultChan() <-chan Event
}

结构体 watchChan 的定义如下：

type watchChan struct {
  watcher           *watcher  // 由那个watcher所生成的
  key               string   // 监听的key
  initialRev        int64    // key的reversion
  recursive         bool     // 是不是要监听key以及key/下的路径
  progressNotify    bool     // 有点keepalive的味道，即每隔一段时间发送一个event，验活
  internalPred      storage.SelectionPredicate
  ctx               context.Context
  cancel            context.CancelFunc
  incomingEventChan chan *event
  resultChan        chan watch.Event
  errChan           chan error
}

watchChan 通过结构体 watcher 调用其方法 Watch来创建：

  • 通过 createWatchChan 来创建 watchChan，并返回。
  • 因为 watchChan 实现了接口 watch.Interface，所以消费者 可以通过 ResultChan 来消费监听到的event了。
func (w *watcher) Watch(ctx context.Context, key string, rev int64, recursive, progressNotify bool, pred storage.SelectionPredicate) (watch.Interface, error) {
	if recursive && !strings.HasSuffix(key, "/") {
		key += "/"
	}
	wc := w.createWatchChan(ctx, key, rev, recursive, progressNotify, pred)
	go wc.run()

	utilflowcontrol.WatchInitialized(ctx)

	return wc, nil
}

watchChan 通过其 run 方法来运行：

func (wc *watchChan) run() {
  watchClosedCh := make(chan struct{})
  go wc.startWatching(watchClosedCh)

  var resultChanWG sync.WaitGroup
  resultChanWG.Add(1)
  go wc.processEvent(&resultChanWG)

  select {
  case err := <-wc.errChan:
    if err == context.Canceled {
      break
    }
    //把错误转化成watch.Event,然后发送到resultChan
    errResult := transformErrorToEvent(err)
    if errResult != nil {
      // error result is guaranteed to be received by user before closing ResultChan.
      select {
      case wc.resultChan <- *errResult:
      case <-wc.ctx.Done(): // user has given up all results
      }
    }
  case <-watchClosedCh:
  case <-wc.ctx.Done(): // user cancel
  }

  // We use wc.ctx to reap all goroutines. Under whatever condition, we should stop them all.
  // It's fine to double cancel.
  wc.cancel()

  // we need to wait until resultChan wouldn't be used anymore
  resultChanWG.Wait()
  close(wc.resultChan)
}

在这里一共启动了2个协程：startWatching 和 processEvent，即监听事件和处理事件。

监听事件

通过 startWatching 来监听事件：

  • 如果 initialRev 提供的 reversion 的值是 0，则首先获取 key 的值，并封装成 event 发送出去，然后在以给定 reversion+1 为 option 中 reversion 的确定值进行监听。
  • 如果 initialRev 提供的 reversion 的值非 0，则给定 reversion+1 为 option 中 reversion 的确定值进行监听。
func (wc *watchChan) startWatching(watchClosedCh chan struct{}) {
	if wc.initialRev == 0 {//如果initialRev为0的话，说明是首次watch，需要做一下同步
		if err := wc.sync(); err != nil {
			klog.Errorf("failed to sync with latest state: %v", err)
			wc.sendError(err)
			return
		}
	}
    //监听当前RV+1
	opts := []clientv3.OpOption{clientv3.WithRev(wc.initialRev + 1), clientv3.WithPrevKV()}
	if wc.recursive {
		opts = append(opts, clientv3.WithPrefix())
	}
	if wc.progressNotify {
		opts = append(opts, clientv3.WithProgressNotify())
	}
    //调用ETCD官方库
	wch := wc.watcher.client.Watch(wc.ctx, wc.key, opts...)
	for wres := range wch {
		if wres.Err() != nil { // 错误处理
			err := wres.Err()
			// If there is an error on server (e.g. compaction), the channel will return it before closed.
			logWatchChannelErr(err) // 日志记录
			wc.sendError(err)//把日志发送到channel
			return
		}
		if wres.IsProgressNotify() {
			wc.sendEvent(progressNotifyEvent(wres.Header.GetRevision()))
			metrics.RecordEtcdBookmark(wc.watcher.objectType)
			continue
		}

        //把 etcd 官方库 event 转化成 event
		for _, e := range wres.Events {
			parsedEvent, err := parseEvent(e)
			if err != nil {
				logWatchChannelErr(err)
				wc.sendError(err)
				return
			}
            //把event发送出去
			wc.sendEvent(parsedEvent)
		}
	}
	// When we come to this point, it's only possible that client side ends the watch.
	// e.g. cancel the context, close the client.
	// If this watch chan is broken and context isn't cancelled, other goroutines will still hang.
	// We should notify the main thread that this goroutine has exited.
	close(watchClosedCh)
}

监听过程产生的 etcd 官方库的 clientv3.Event 需要转换成内部的 event 通过 sendEvent 发送出去。 sendEvent 的实现很简单，从逻辑来看是一个阻塞发送，当 channel 中的元素数量为 incomingBufSize 的时候，会打印 warnning 日志

func (wc *watchChan) sendEvent(e *event) {
	if len(wc.incomingEventChan) == incomingBufSize {
		klog.V(3).InfoS("Fast watcher, slow processing. Probably caused by slow decoding, user not receiving fast, or other processing logic", "incomingEvents", incomingBufSize, "objectType", wc.watcher.objectType)
	}
	select {
	case wc.incomingEventChan <- e:
	case <-wc.ctx.Done():
	}
}

处理事件

processEvent 的逻辑也很明了， 从管道中获取 event， 然后转换成 watch.Event 推送到管道供监听者消费：

// processEvent processes events from etcd watcher and sends results to resultChan.
func (wc *watchChan) processEvent(wg *sync.WaitGroup) {
  defer wg.Done()

  for {
    select {
    case e := <-wc.incomingEventChan://获取到event
      // 把event转换为watch.Event
      res := wc.transform(e)
      if res == nil {
        continue
      }
      if len(wc.resultChan) == outgoingBufSize {
        klog.V(3).InfoS("Fast watcher, slow processing. Probably caused by slow dispatching events to watchers", "outgoingEvents", outgoingBufSize, "objectType", wc.watcher.objectType)
      }
      //把watch.event发送到resultChan中
      select {
      case wc.resultChan <- *res:
      case <-wc.ctx.Done():
        return
      }
    case <-wc.ctx.Done():
      return
    }
  }
}
